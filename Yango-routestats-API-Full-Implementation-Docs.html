POST https://tc.taxi.yango.tech/3.0/routestats 200 OK

Fetches available tariff classes, prices, and ETAs for a given route. Called before placing an order to display ride options to the user.

Constants & Variables

Store these in your environment config or .env file. Values marked REQUIRED must be set before calling the API.

Implementation Sample (Node.js)

// ─── CONFIG / CONSTANTS ───────────────────────────────────────────
const BASE_URL        = "https://tc.taxi.yango.tech";
const BEARER_TOKEN    = process.env.YANGO_BEARER_TOKEN; // from vault/env

// Device identifiers (stable per install)
const YANGO_USER_ID        = "6fd050108263479594457f48f91b8fac";
const APPMETRICA_UUID      = "2f4fa6cdb3fb4348b9b857b88aee96ae";
const APPMETRICA_DEVICE_ID = "7839ae96fe47e50b34aadc7cf31f3338";
const MOB_ID               = "9b18ee6380f945e7a39bc5ea44c37cda";

// App constants
const APP_VERSION    = "5.98.2.128633";
const APP_BUILD      = "50128633";
const ANDROID_VER    = "10";
const DEVICE_MODEL   = "samsung; Nexus";
const PERF_CLASS     = "medium";
const ACCEPT_LANG    = "en-US";
const MOBCF          = "yango%25yango_not_eu_by_geo_7%25yango_forced_ac";
const MOBPR          = "yango_not_eu_by_geo_7_TAXI_0";

// ─── FUNCTION ─────────────────────────────────────────────────────
async function getRouteStats({
  sessionId,        // generate per session (uuid)
  zoneName,         // e.g. "lahore"
  deviceLat, deviceLon,
  pickupLat, pickupLon, pickupTitle,
  dropoffLat, dropoffLon, dropoffTitle,
  selectedClass = "econom",
  paymentType   = "cash",
}) {
  const url = `${BASE_URL}/3.0/routestats?mobcf=${MOBCF}&mobpr=${MOBPR}`;

  const headers = {
    "Host":                              "tc.taxi.yango.tech",
    "Content-Type":                      "application/json; charset=utf-8",
    "Authorization":                     `Bearer ${BEARER_TOKEN}`,
    "X-Oauth-Token":                     BEARER_TOKEN,
    "X-Location":                        `${deviceLat};${deviceLon}`,
    "X-Yataxi-Userid":                   YANGO_USER_ID,
    "X-Yataxi-Last-Zone-Names":          zoneName,
    "X-Yataxi-Has-Ongoing-Orders":       "false",
    "X-Yataxi-Ongoing-Orders-Statuses":  "none",
    "X-Ya-Go-Superapp-Session":          sessionId,
    "X-Appmetrica-Uuid":                 APPMETRICA_UUID,
    "X-Appmetrica-Deviceid":             APPMETRICA_DEVICE_ID,
    "X-Mob-Id":                          MOB_ID,
    "User-Agent":   `yango/${APP_VERSION} Android/${ANDROID_VER} (${DEVICE_MODEL})`,
    "X-User-Agent-Split": `yango ${APP_VERSION.slice(0,6)} (${APP_BUILD})`,
    "X-Perf-Class":  PERF_CLASS,
    "Accept-Language": ACCEPT_LANG,
    "X-Vpn-Active":  "0",
  };

  const body = {
    id:               YANGO_USER_ID,
    zone_name:        zoneName,
    selected_class:   selectedClass,
    route: [
      [pickupLon,  pickupLat],   // ⚠️ longitude FIRST
      [dropoffLon, dropoffLat],
    ],
    tariff_requirements: [
      "business","cargo","comfortplus","courier","econom",
      "express","intercity","intercity_preorder","moto",
    ].map(c => ({ class: c })),
    force_soon_order:        false,
    payment:                 { type: paymentType },
    skip_estimated_waiting:  false,
    size_hint:               352,
    multiclass_options:      { class: [], verticals: [], selected: false },
    use_toll_roads:          false,
    is_lightweight:          false,
    supported_vertical_types: ["group"],
    supported_verticals: ["transport","intercity","taxi","delivery","rest_tariffs"],
    state: {
      fields: [
        { type: "a", title: pickupTitle,  entrance: "",
          position: [pickupLon,  pickupLat],  log: "" },
        { type: "b", title: dropoffTitle, entrance: "",
          position: [dropoffLon, dropoffLat], log: "" },
      ],
      location: [deviceLon, deviceLat],
      precise_location_available: true,
    },
    suggest_alternatives:           true,
    extended_description:           true,
    with_title:                     true,
    supported_markup:               "tml-0.1",
    selected_class_only:            false,
    supports_hideable_tariffs:      true,
    summary_version:                2,
    supports_unavailable_alternatives: true,
    supports_verticals_selector:    true,
    supports_no_cars_available:     true,
    supports_paid_options:          true,
    format_currency:                true,
    supports_explicit_antisurge:    true,
    supports_multiclass:            true,
    account_type:                   "lite",
  };

  const res = await fetch(url, {
    method:  "POST",
    headers,
    body:    JSON.stringify(body),
  });

  if (!res.ok) throw new Error(`routestats failed: ${res.status}`);
  return res.json(); // { currency_rules, service_levels: [...] }
}

// ─── USAGE ────────────────────────────────────────────────────────
const data = await getRouteStats({
  sessionId:    crypto.randomUUID().replace(/-/g,""),
  zoneName:     "lahore",
  deviceLat:    31.47424733, deviceLon: 74.41635884,
  pickupLat:    31.474148968180497, pickupLon: 74.41626832907782,
  pickupTitle:  "Haider Street, 29/1",
  dropoffLat:   31.118998, dropoffLon: 74.45399,
  dropoffTitle: "Destination",
});

console.log(data.service_levels.map(s =>
  `${s.name}: ${s.price}`
));
